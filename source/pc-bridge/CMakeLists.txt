cmake_minimum_required(VERSION 3.16)
project(frame_pc_bridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  add_compile_options(/W4 /WX /EHsc)
  add_compile_definitions(_CRT_SECURE_NO_WARNINGS NOMINMAX)
  if(BRIDGE_ASAN)
    add_compile_options(/fsanitize=address /Zi)
    add_link_options(/INCREMENTAL:NO /DEBUG)
  endif()
else()
  add_compile_options(-Wall -Wextra -Wpedantic -Werror)
  if(BRIDGE_ASAN)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
  endif()
endif()
include_directories(../third_party/openvr)
add_executable(pico_pc_bridge_send send.cpp)
add_executable(pc_bridge_receive receive.cpp)
add_executable(pc_bridge_tests tests.cpp)
if(WIN32)
  target_link_libraries(pico_pc_bridge_send PRIVATE ws2_32)
  target_link_libraries(pc_bridge_receive PRIVATE ws2_32)
  add_library(driver_frametrackers SHARED driver.cpp)
  target_link_libraries(driver_frametrackers PRIVATE ws2_32)
  set_target_properties(driver_frametrackers PROPERTIES PREFIX "")
  # Staging stays in build tree; never modifies an installed SteamVR driver.
  add_custom_command(TARGET driver_frametrackers POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/driver" "$<TARGET_FILE_DIR:driver_frametrackers>/frametrackers"
    COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:driver_frametrackers>/frametrackers/bin/win64"
    COMMAND ${CMAKE_COMMAND} -E copy "$<TARGET_FILE:driver_frametrackers>" "$<TARGET_FILE_DIR:driver_frametrackers>/frametrackers/bin/win64/driver_frametrackers.dll")
else()
  target_link_libraries(pico_pc_bridge_send PRIVATE dl pthread)
endif()
enable_testing()
add_test(NAME pc_bridge_tests COMMAND pc_bridge_tests)
